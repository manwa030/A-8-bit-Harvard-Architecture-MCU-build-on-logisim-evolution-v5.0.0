;Note: li r7, anyNums =addr111 wriEN1 ,so, if enter =1, li key board data to r7

main:
        li   r7, 0     ;listen keyboard
        beq  r7, r0, main    ; no key
        mv   r6, r7          ; r6 = keyboard value 0..255

        ; ---- clear digits ----
        li   r1, 0           ; hundreds
        li   r2, 0           ; tens
        li   r3, 0           ; ones

        ; ---- hundreds ----
        li   r4, 100
hundreds_loop:
        bltu r6, r4, hundreds_done
        sub  r6, r6, r4
        addi r1, r1, 1
        j    hundreds_loop
hundreds_done:

        ; ---- tens ----
        li   r4, 10
tens_loop:
        bltu r6, r4, tens_done
        sub  r6, r6, r4
        addi r2, r2, 1
        j    tens_loop
tens_done:

        ; ---- ones ----
        mv   r3, r6

        ; ---- output 3 digits ----
        sw   r1, 255(r0)     ; hundreds
        sw   r2, 255(r0)     ; tens
        sw   r3, 255(r0)     ; ones

        li   r6, 0
        j    main
